package transaction.cashin.TransactionService.Integration.Amdocs.Authentication;

import org.apache.camel.builder.RouteBuilder;
import org.springframework.stereotype.Component;

import java.net.URLEncoder;
import java.nio.charset.StandardCharsets;

import org.apache.camel.builder.RouteBuilder;
import org.springframework.stereotype.Component;

@Component
public class AuthenticationRoute extends RouteBuilder {

    @Override
    public void configure() {

        from("direct:amdocs-authentication")
            .routeId("amdocs-authentication-route")

            .setHeader("Content-Type")
                .constant(
                    "application/x-www-form-urlencoded")

            .process(exchange -> {

                AuthenticationRequest request =
                        exchange.getMessage()
                                .getBody(
                                    AuthenticationRequest.class);

                String formBody =
                        "grant_type="
                        + encode(request.getGrantType())
                        + "&api_key="
                        + encode(request.getApiKey())
                        + "&password="
                        + encode(request.getPassword())
                        + "&username="
                        + encode(request.getUsername());

                exchange.getMessage()
                        .setBody(formBody);
            })

            .to("{{amdocs.authentication.url}}?httpMethod=POST")

            .unmarshal()
                .json(AuthenticationResponse.class);
    }

    private String encode(String value) {

        return URLEncoder.encode(
                value == null ? "" : value,
                StandardCharsets.UTF_8);
    }
}

@Component
public class GroupRoute extends RouteBuilder {

    @Value("${amdocs.client-id}")
    private String clientId;

    @Value("${amdocs.api-key}")
    private String apiKey;

    @Value("${amdocs.source-channel:MobileApp}")
    private String sourceChannel;

    @Value("${amdocs.wait:11}")
    private String wait;

    @Value("${amdocs.suppress-sms:0}")
    private String suppressSms;
    
    
    
    
    @Bean("dateHeaderFilterStrategy")
    public HeaderFilterStrategy dateHeaderFilterStrategy() {
        return new HeaderFilterStrategy() {

            @Override
            public boolean applyFilterToCamelHeaders(
                    String headerName,
                    Object headerValue,
                    org.apache.camel.Exchange exchange) {

                return false;
            }

            @Override
            public boolean applyFilterToExternalHeaders(
                    String headerName,
                    Object headerValue,
                    org.apache.camel.Exchange exchange) {

                return false;
            }
        };
    }
    
    
    

    @Override
    public void configure() {

        from("direct:amdocs-get-group")
            .routeId("amdocs-get-group-route")

            // =====================================================
            // Validate and build account group path
            // =====================================================

            .process(exchange -> {

                GroupRequest request =
                        exchange.getMessage()
                                .getBody(GroupRequest.class);

                if (request == null ||
                        request.getReference() == null ||
                        request.getReference().isBlank()) {

                    throw new IllegalArgumentException(
                            "Group account reference is required");
                }

                String groupPath =
                        "accountid@"
                        + request.getReference()
                        + "/groups";

                exchange.setProperty(
                        "groupPath",
                        groupPath);
            })

            // =====================================================
            // Common Amdocs headers
            // =====================================================

            .process(exchange ->
            exchange.getMessage().setHeader(
                "Date",
                Instant.now().toString()
            )
        )

            .setHeader("X-Client-ID")
                .constant(clientId)

            .setHeader("X-API-Key")
                .constant(apiKey)

            .process(exchange ->
                exchange.getMessage().setHeader(
                        "x-correlationid",
                        UUID.randomUUID().toString()
                )
            )

            .setHeader("source_channel")
                .constant(sourceChannel)

            .setHeader("wait")
                .constant(wait)

            .setHeader("suppress_sms")
                .constant(suppressSms)

            .setHeader("Content-Type")
                .constant("application/json")

            // =====================================================
            // GET Group API
            // =====================================================
                
            .log("Date header before HTTP call: ${header.Date}")
            .log("DFS Group request body: ${body}")


            .setHeader("CamelHttpMethod")
                .constant("GET")

            .toD(
                "{{amdocs.group.url}}/"
                + "${exchangeProperty.groupPath}"
                + "?httpMethod=GET"
                + "&headerFilterStrategy=#dateHeaderFilterStrategy"
            )
            
            
            .log("DFS Group HTTP status: ${header.CamelHttpResponseCode}")
            .log("DFS Group response body: ${body}")

            // =====================================================
            // Deserialize response
            // =====================================================

            .unmarshal()
                .json(GroupResponse.class);
    }
}

@Service
public class TransactionService {

    private final PartyRepository partyRepository;

    private final TransactionRepository transactionRepository;

    private final TransactionValidator transactionValidator;

    private final CashInValidator cashInValidator;

    private final QuoteService quoteService;

    private final GroupService groupService;

    private final CashInService cashInService;

    private final AuthenticationService authenticationService;


    public TransactionService(

            PartyRepository partyRepository,

            TransactionRepository transactionRepository,

            TransactionValidator transactionValidator,

            CashInValidator cashInValidator,

            QuoteService quoteService,

            GroupService groupService,

            CashInService cashInService,

            AuthenticationService authenticationService) {

        this.partyRepository = partyRepository;

        this.transactionRepository = transactionRepository;

        this.transactionValidator = transactionValidator;

        this.cashInValidator = cashInValidator;

        this.quoteService = quoteService;

        this.groupService = groupService;

        this.cashInService = cashInService;

        this.authenticationService = authenticationService;
    }


    @Transactional
    public TransactionResponse createTransaction(
            TransactionRequest request) {

       
        // 1. Common validation
        

        transactionValidator.validate(request);


       
        // 2. Cash-In specific validation
       

        cashInValidator.validate(request);


        
        // 3. Authenticate with Amdocs
       

        AuthenticationResponse authenticationResponse =
                authenticationService.authenticate();

        String accessToken =
                authenticationResponse.getAccessToken();


        if (accessToken == null ||
                accessToken.isBlank()) {

            throw new IllegalStateException(
                    "Amdocs authentication failed: no access token returned");
        }


       
        // 4. Resolve parties
 

      //  Party initiator =
        //        findPartyByMsisdn(
        //                request.getInitiatorMsisdn());

      //  Party receiver =
       //         findPartyByMsisdn(
       //                 request.getReceiverMsisdn());


        // 5. Check initiator Amdocs groups
  
        GroupRequest initiatorGroupRequest =
                new GroupRequest(
                        request.getInitiatorMsisdn());


        GroupResponse initiatorGroupResponse =
                groupService.getGroup(
                        initiatorGroupRequest,
                        accessToken);


        if (initiatorGroupResponse == null) {

            throw new IllegalStateException(
                    "Amdocs Group API returned no response " +
                    "for initiator");
        }


        /*
        //Please return this as soon as you're done testing. It's super neccessary
        if (!hasGroup(
                initiatorGroupResponse,
                "RET_Assujetti")) {

            throw new IllegalStateException(
                    "Initiator is not a member of the " +
                    "RET_Assujetti group");
        }
		*/
        
        

        //
        // 6. Check receiver Amdocs groups
        //
  

        GroupRequest receiverGroupRequest =
                new GroupRequest(
                        request.getReceiverMsisdn());


        GroupResponse receiverGroupResponse =
                groupService.getGroup(
                        receiverGroupRequest,
                        accessToken);


        if (receiverGroupResponse == null) {

            throw new IllegalStateException(
                    "Amdocs Group API returned no response " +
                    "for receiver");
        }


        if (!hasGroup(
                receiverGroupResponse,
                "Customers")) {

            throw new IllegalStateException(
                    "Receiver is not a member of the " +
                    "Customers group");
        }



        // 7. Build Quote API request
       

        String quoteInitiator =
                request.getInitiatorMsisdn();


        QuoteRequest quoteRequest =
                new QuoteRequest();


        quoteRequest.setRequestAmount(
                request.getAmount().toPlainString());


        quoteRequest.setRequestCurrency(
                request.getCurrency());


        quoteRequest.setRequestDate(
                Instant.now().toString());


        // ---------------------------------------------------------
        // Debit party
        // ---------------------------------------------------------

        quoteRequest.setDebitParty(
                List.of(
                        new QuoteParty(
                                "accountid",
                                request.getInitiatorMsisdn()
                        )
                )
        );


        // ---------------------------------------------------------
        // Credit party
        // ---------------------------------------------------------

        quoteRequest.setCreditParty(
                List.of(
                        new QuoteParty(
                                "accountid",
                                request.getReceiverMsisdn()
                        )
                )
        );


        // ---------------------------------------------------------
        // Quote metadata
        // ---------------------------------------------------------

        quoteRequest.setMetadata(
                List.of(

                        new QuoteMetadata(
                                "description",
                                request.getReference()
                        ),

                        new QuoteMetadata(
                                "subType",
                                request.getSubType()
                        )
                )
        );


        // =========================================================
        // 8. Call Amdocs Quote API
        // =========================================================

        QuoteResponse quoteResponse =
                quoteService.getQuote(
                        quoteRequest,
                        accessToken,
                        quoteInitiator);


        // =========================================================
        // 9. Validate Quote response
        // =========================================================

        if (quoteResponse == null) {

            throw new IllegalStateException(
                    "Amdocs Quote API returned no response");
        }


        if (quoteResponse.getQuotationReference() == null ||
                quoteResponse.getQuotationReference().isBlank()) {

            throw new IllegalStateException(
                    "Amdocs Quote API returned no quotation reference");
        }


        // =========================================================
        // 10. Build Cash-In request
        // =========================================================

        CashInRequest cashInRequest =
                new CashInRequest();


        cashInRequest.setAmount(
                request.getAmount().toPlainString());


        cashInRequest.setCurrency(
                request.getCurrency());


        cashInRequest.setSubType(
                request.getSubType());


        cashInRequest.setType(
                "transfer");


        cashInRequest.setRequestDate(
                Instant.now().toString());


        // ---------------------------------------------------------
        // Debit party
        //
        // Amdocs Cash-In example:
        // accountid = mvola
        // ---------------------------------------------------------

        CashInParty debitParty =
                new CashInParty(
                        "accountid",
                        "mvola");


        // ---------------------------------------------------------
        // Credit party
        // ---------------------------------------------------------

        CashInParty creditParty =
                new CashInParty(
                        "accountid",
                        request.getReceiverMsisdn());


        cashInRequest.setDebitParty(
                List.of(debitParty));


        cashInRequest.setCreditParty(
                List.of(creditParty));


        // ---------------------------------------------------------
        // Cash-In metadata
        // ---------------------------------------------------------

        CashInMetadata description =
                new CashInMetadata(
                        "description",
                        request.getReference());


        cashInRequest.setMetadata(
                List.of(description));


        // =========================================================
        // 11. Execute Cash-In
        // =========================================================

        CashInResponse cashInResponse =
                cashInService.executeCashIn(
                        cashInRequest,
                        accessToken);


        // =========================================================
        // 12. Validate Cash-In response
        // =========================================================

        if (cashInResponse == null) {

            throw new IllegalStateException(
                    "Amdocs Cash-In API returned no response");
        }


        if (cashInResponse.getTransactionStatus() == null ||
                !"completed".equalsIgnoreCase(
                        cashInResponse.getTransactionStatus())) {

            throw new IllegalStateException(
                    "Amdocs Cash-In operation failed. " +
                    "Transaction status: "
                    + cashInResponse.getTransactionStatus());
        }


        // =========================================================
        // 13. Create Cash-In transaction
        // =========================================================

        CashInTransaction transaction =
                new CashInTransaction();


        transaction.setAmount(
                request.getAmount());


        transaction.setCurrency(
                request.getCurrency());


        transaction.setSubType(
                request.getSubType());


        transaction.setReference(
                request.getReference());


        transaction.setStatus(
                cashInResponse.getTransactionStatus());


        transaction.setCreatedAt(
                LocalDateTime.now());


     //   transaction.setInitiator(
      //          initiator);


      //  transaction.setReceiver(
       //         receiver);


        // =========================================================
        // 14. Store Amdocs quotation information
        // =========================================================

        transaction.setQuotationReference(
                quoteResponse.getQuotationReference());


        transaction.setQuotationReceipt(
                quoteResponse.getQuotationReceipt());


        // =========================================================
        // 15. Persist transaction
        // =========================================================

        CashInTransaction savedTransaction =
                (CashInTransaction)
                        transactionRepository.save(
                                transaction);


        // =========================================================
        // 16. Return REST response
        // =========================================================

        return new TransactionResponse(

                savedTransaction.getId(),

                request.getTransactionType(),

                savedTransaction.getAmount(),

                savedTransaction.getCurrency(),

                savedTransaction.getStatus(),

                savedTransaction.getReference(),

                savedTransaction.getQuotationReference(),

                savedTransaction.getQuotationReceipt()
        );
    }


    // =============================================================
    // Check whether a group response contains a specific group
    // =============================================================

    private boolean hasGroup(
            GroupResponse groupResponse,
            String groupName) {

        if (groupResponse.getAgentGroups() == null) {
            return false;
        }


        return groupResponse.getAgentGroups()
                .stream()
                .map(AgentGroup::getName)
                .anyMatch(
                        name -> name != null &&
                                name.equalsIgnoreCase(groupName)
                );
    }


    // =============================================================
    // Find Party
    // =============================================================

    private Party findPartyByMsisdn(
            String msisdn) {

        return partyRepository
                .findByMsisdn(msisdn)
                .orElseThrow(
                        () -> new PartyNotFoundException(msisdn)
                );
    }
}
